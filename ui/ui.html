// server.js — Single-file UI + proxy to avoid CORS // Usage: // 1) Ensure your KServe port-forward is running locally, e.g.: // kubectl -n default port-forward svc/distilgpt2-predictor 8085:80 // 2) In this folder, run: // npm init -y // npm i express http-proxy-middleware // node server.js // 3) Open http://localhost:3000 in your browser. // 4) In the UI, click "Self‑test". You should see your model list. // // Why this fixes "TypeError: Failed to fetch" + CORS errors: // Opening an HTML file directly (origin "null") or from a different // port triggers CORS. This tiny Express app serves the UI and proxies // all /openai/* calls to your KServe server on http://localhost:8085, // keeping everything on the same origin (http://localhost:3000), so // the browser no longer blocks requests. const express = require('express'); const path = require('path'); const { createProxyMiddleware } = require('http-proxy-middleware'); // Target KServe model server (change if your port-forward differs) const KSERVE_URL = process.env.KSERVE_URL || 'http://localhost:8085'; const PORT = process.env.PORT || 3000; const app = express(); // --- Proxy: same-origin bridge to KServe OpenAI shim --- // When the UI calls /openai/v1/... it will be forwarded to // ${KSERVE_URL}/openai/v1/... app.use('/openai', createProxyMiddleware({ target: KSERVE_URL, changeOrigin: true, ws: true, // Do NOT rewrite the path; KServe expects the /openai prefix })); // --- Minimal UI (served from /) --- const INDEX_HTML = ` KServe Chat UI (CORS-safe)

KServe Chat UI

Proxy → ${KSERVE_URL}
Troubleshooting
  1. Ensure port-forward is running: kubectl -n default port-forward svc/distilgpt2-predictor 8085:80
  2. This app proxies /openai/* to ${KSERVE_URL} to avoid CORS. Calls go to:
    • /openai/v1/completions (for non-chat)
    • /openai/v1/chat/completions (for chat)
  3. If you see "chat template is no longer allowed", uncheck Use Chat and send as a plain completion.
`; app.get('/', (_req, res) => res.status(200).type('html').send(INDEX_HTML)); app.listen(PORT, () => { console.log(`UI listening on http://localhost:${PORT}`); console.log(`Proxying /openai/* → ${KSERVE_URL}`); });
